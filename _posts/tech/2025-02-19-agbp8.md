#### 手机端模式切换
> 原理：网站是电脑端访问是电脑端，手机端访问主域名会自动跳转链接到手机端网址，通过加cookie判断电脑端访问是否自动跳转到手机端

```js
<script src="/assets/default/js/uaredirect.js" type="text/javascript"></script>
<script type="text/javascript">
  $(".mini_menu a:eq(0)").on("click", function () {
    document.cookie = "m2wcookie=1;path=/";
  });
  $(".mini_menu a:eq(1)").on("click", function () {
    document.cookie = "m2wcookie=0;path=/";
  });
  var arrStr = document.cookie.split("; ");
  let is_redirect = 0;
  for (var i = 0; i < arrStr.length; i++) {
    var temp = arrStr[i].split("=");
    if (temp[0] == 'm2wcookie') {
      is_redirect = unescape(temp[1]);
    }
  }
  if (is_redirect != 1) {
    uaredirect("/wap_new.php");
  }
</script>
```
